## Store the tween

All tween methods return a **Tween** instance:

```tsx
const tween = gsap.to('.box', { x: 100, duration: 1, paused: true });
tween.play();
tween.pause();
tween.reverse();
tween.progress(0.5);
tween.kill();
```

## Hover reverse (`contextSafe`)

<ComponentPreview>
  <GsapHoverReverse />

  <DocsFileTabs title="Hover reverse">

```tsx title="hover-reverse.tsx"
const onEnter = contextSafe(() => {
  gsap.to('.card', { y: -6, scale: 1.02, duration: 0.2, ease: 'power2.out' });
});
const onLeave = contextSafe(() => {
  gsap.to('.card', { y: 0, scale: 1, duration: 0.2, ease: 'power2.out' });
});
```

```html title="hover-reverse-markup.html"
<div class="gsap-card-lift w-48 rounded-xl border bg-card p-4 shadow-sm">
  <p>Hover me</p>
  <p>Reverses on leave</p>
</div>
```
