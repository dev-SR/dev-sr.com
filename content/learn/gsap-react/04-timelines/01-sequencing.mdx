## Modal open sequence

Overlay → panel → staggered children with `"<"` and `"-="`:

<ComponentPreview>
  <GsapModalSequence />

  <DocsFileTabs title="Modal timeline">

```tsx title="modal-timeline.tsx"
const tl = gsap.timeline({ defaults: { ease: 'power2.out' } });
tl.from('.overlay', { autoAlpha: 0, duration: 0.2 })
  .from('.modal', { scale: 0.95, autoAlpha: 0, duration: 0.28 }, '<0.05')
  .from('.modal-item', { y: 12, autoAlpha: 0, stagger: 0.06, duration: 0.22 }, '-=0.1');
```

```html title="modal-timeline-markup.html"
<div class="absolute inset-0 flex items-center justify-center">
  <div class="gsap-overlay absolute inset-0 bg-black/40"></div>
  <div class="gsap-modal relative w-64 rounded-xl border bg-card p-4">
    <p class="gsap-modal-item">Delete project?</p>
    <p class="gsap-modal-item">This cannot be undone.</p>
    <div class="gsap-modal-item flex gap-2">
      <button>Cancel</button>
      <button>Delete</button>
    </div>
  </div>
</div>
```

## Splash-style letters

<ComponentPreview replay>
  <GsapSplashLetters />

  <DocsFileTabs title="Splash letters">

```tsx title="splash-letters.tsx"
gsap.from('.letter', {
  y: 24,
  autoAlpha: 0,
  rotation: 8,
  stagger: 0.04,
  duration: 0.45,
  ease: 'back.out(1.4)',
});
```

