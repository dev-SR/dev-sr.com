## quickTo — mouse follower

Reuse one tween instead of creating hundreds per frame:

<ComponentPreview>
  <GsapQuickToFollower />

  <DocsFileTabs title="Quick to">

```tsx title="quick-to.tsx"
const xTo = gsap.quickTo('.follower', 'x', { duration: 0.4, ease: 'power3' });
const yTo = gsap.quickTo('.follower', 'y', { duration: 0.4, ease: 'power3' });
xTo(gsap.utils.clamp(0, maxX, x));
yTo(gsap.utils.clamp(0, maxY, y));
```

```html title="quick-to-markup.html"
<div class="relative h-full cursor-crosshair">
  <div class="gsap-follower absolute size-8 rounded-full"></div>
</div>
```

  </DocsFileTabs>
</ComponentPreview>

## random stagger

<ComponentPreview replay>
  <GsapRandomStagger />

  <DocsFileTabs title="Random stagger">

```tsx title="random-stagger.tsx"
gsap.to('.dot', {
  x: 'random(-40, 40, 5)',
  y: 'random(-20, 20, 5)',
  stagger: { amount: 0.3, from: 'random' },
});
```

```html title="random-stagger-markup.html"

  <!-- …more .gsap-rand-dot -->
```
