## Structured logging patterns

Use consistent log templates so the dashboard filter bar stays useful across services.

```csharp title="OrderProcessor/Workers/OrderConsumer.cs"
public sealed class OrderConsumer(
    ILogger<OrderConsumer> logger,
    CatalogApiClient catalog,
    IOrderRepository orders)
{
    public async Task HandleAsync(OrderPlaced message, CancellationToken ct)
    {
        using var scope = logger.BeginScope(new Dictionary<string, object>
        {
            ["OrderId"] = message.OrderId,
            ["CustomerId"] = message.CustomerId,
        });

        logger.LogInformation("Processing order {OrderId} with {LineCount} lines", message.OrderId, message.Lines.Count);

        foreach (var line in message.Lines)
        {
            logger.LogDebug("Checking stock for SKU {Sku}", line.Sku);
            var available = await catalog.CheckStockAsync(line.Sku, ct);
            if (available < line.Quantity)
            {
                logger.LogWarning("Insufficient stock for {Sku}: need {Need}, have {Have}", line.Sku, line.Quantity, available);
                await orders.MarkFailedAsync(message.OrderId, "Insufficient stock", ct);
                return;
            }
        }

        await orders.MarkCompletedAsync(message.OrderId, ct);
        logger.LogInformation("Order {OrderId} completed", message.OrderId);
    }
}
```

```tsx title="components/resource-status.tsx"
import { Badge } from '@/components/ui/badge';

type ResourceState = 'Running' | 'Waiting' | 'Stopped';

export function ResourceStatus({ state }: { state: ResourceState }) {
  const variant =
    state === 'Running' ? 'default' : state === 'Waiting' ? 'secondary' : 'outline';

  return <Badge variant={variant}>{state}</Badge>;
}
```

## Troubleshooting checklist

<Steps>
  <Step>Confirm the AppHost is the startup project</Step>

  ```bash
  dotnet run --project AppHost
  ```

  <Step>Open the dashboard URL printed in the console</Step>

  Usually `https://localhost:17xxx` — accept the dev certificate if prompted.

  <Step>Verify OTLP endpoints match between AppHost and services</Step>

  ```bash
  echo $DOTNET_DASHBOARD_OTLP_ENDPOINT_URL
  ```

  <Step>Check `/health` returns 200 for each API resource</Step>
</Steps>
