## Single large fence

Use a normal fenced block with `title="path/file.ext"`. Blocks longer than ~13 lines show **Expand** with a gradient fade.

```tsx title="components/learn/docs-sidebar.tsx"
'use client';

import { ChevronRight } from 'lucide-react';
import Link from 'next/link';
import { usePathname } from 'next/navigation';
import { Accordion, AccordionContent, AccordionItem, AccordionTrigger } from '@/components/ui/accordion';
import { ScrollArea } from '@/components/ui/scroll-area';
import { cn } from '@/lib/utils';
import type { LearnNavNode } from '@/lib/learn';

function NavLink({ node, depth = 0 }: { node: LearnNavNode; depth?: number }) {
  const pathname = usePathname();
  const active = pathname === node.href;

  return (
    <Link
      href={node.href}
      className={cn(
        'flex items-center gap-2 rounded-md px-2 py-1.5 text-sm transition-colors',
        active ? 'bg-muted text-foreground' : 'text-muted-foreground hover:text-foreground'
      )}
      style={{ paddingLeft: `${8 + depth * 12}px` }}>
      {node.title}
      {node.badge && (
        <span className="ml-auto rounded-full bg-primary/10 px-1.5 py-0.5 text-[0.65rem] font-medium text-primary">
          {node.badge}
        </span>
      )}
    </Link>
  );
}

function NavSection({ node, depth = 0 }: { node: LearnNavNode; depth?: number }) {
  if (node.type === 'page' || !node.children?.length) {
    return <NavLink node={node} depth={depth} />;
  }

  return (
    <Accordion type="multiple" defaultValue={[node.slug]}>
      <AccordionItem value={node.slug} className="border-0">
        <AccordionTrigger className="py-1.5 text-sm hover:no-underline">
          {node.title}
        </AccordionTrigger>
        <AccordionContent className="pb-0 pl-2">
          {node.children.map((child) => (
            <NavSection key={child.slug} node={child} depth={depth + 1} />
          ))}
        </AccordionContent>
      </AccordionItem>
    </Accordion>
  );
}

export function DocsSidebar({ nav }: { nav: LearnNavNode[] }) {
  return (
    <ScrollArea className="h-full pr-3">
      <nav className="space-y-1">
        {nav.map((node) => (
          <NavSection key={node.slug} node={node} />
        ))}
      </nav>
    </ScrollArea>
  );
}
```

```tsx title="components/learn/learn-code-elements.tsx"
export function LearnPreCustom(props: LearnPreProps) {
  const inPrettyFigure = useInDocsPrettyFigure();
  const language = String(props['data-language'] || 'text').toLowerCase();

  if (inPrettyFigure) {
    return <pre className="code-pre m-0" {...props} />;
  }

  return (
    <CodeBlock language={language} rawString={props.__rawstring__}>
      <pre className="code-pre m-0" {...props} />
    </CodeBlock>
  );
}
```
