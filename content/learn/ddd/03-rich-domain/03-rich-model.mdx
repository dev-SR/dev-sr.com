Static in-memory dictionaries remain for teaching. Notice: IDs are immutable, collections are read-only outward, and mutations go through methods.

## Customer

```csharp title="Domain/Customer.cs"
using System.Collections.Concurrent;

namespace RentalDddDemo.Domain;

public sealed class Customer
{
    private static readonly ConcurrentDictionary<string, Customer> Customers = new();
    private readonly List<Rental> _rentals = new();

    static Customer()
    {
        var john = new Customer("customer-1", "John Doe");
        Customers.TryAdd(john.Id, john);
    }

    private Customer(string id, string name)
    {
        if (string.IsNullOrWhiteSpace(name))
        {
            throw new ArgumentException("Name cannot be null or empty.", nameof(name));
        }

        Id = id;
        Name = name;
    }

    public string Id { get; }

    public string Name { get; private set; }

    public IReadOnlyCollection<Rental> Rentals => _rentals.AsReadOnly();

    public static Customer? FindById(string id)
    {
        return Customers.TryGetValue(id, out var customer) ? customer : null;
    }

    public void UpdateContactInformation(string newName)
    {
        if (string.IsNullOrWhiteSpace(newName))
        {
            throw new ArgumentException("Name cannot be null or empty.", nameof(newName));
        }

        Name = newName;
    }

    internal void AddRental(Rental rental)
    {
        if (rental is null)
        {
            throw new ArgumentNullException(nameof(rental));
        }

        if (_rentals.Any(existing => existing.Id == rental.Id))
        {
            return;
        }

        _rentals.Add(rental);
    }
}
```

## Equipment

```csharp title="Domain/Equipment.cs"
using System.Collections.Concurrent;

namespace RentalDddDemo.Domain;

public sealed class Equipment
{
    private static readonly ConcurrentDictionary<string, Equipment> Items = new();

    static Equipment()
    {
        var hammer = new Equipment("equipment-1", "Hammer");
        Items.TryAdd(hammer.Id, hammer);
    }

    private Equipment(string id, string name)
    {
        if (string.IsNullOrWhiteSpace(name))
        {
            throw new ArgumentException("Name cannot be null or empty.", nameof(name));
        }

        Id = id;
        Name = name;
        IsAvailable = true;
    }

    public string Id { get; }

    public string Name { get; private set; }

    public bool IsAvailable { get; private set; }

    public static Equipment? FindById(string id)
    {
        return Items.TryGetValue(id, out var equipment) ? equipment : null;
    }

    public void UpdateName(string newName)
    {
        if (string.IsNullOrWhiteSpace(newName))
        {
            throw new ArgumentException("Name cannot be null or empty.", nameof(newName));
        }

        Name = newName;
    }

    public void Reserve()
    {
        if (!IsAvailable)
        {
            throw new InvalidOperationException("Equipment is not available.");
        }

        IsAvailable = false;
    }

    public void Release()
    {
        IsAvailable = true;
    }
}
```

## Rental

```csharp title="Domain/Rental.cs"
using System.Collections.Concurrent;

namespace RentalDddDemo.Domain;

public sealed class Rental
{
    private static readonly ConcurrentDictionary<string, Rental> Rentals = new();
    private static readonly decimal CostPerDay = 25.00m;

    private Rental(
        Customer customer,
        Equipment equipment,
        DateOnly startDate,
        DateOnly endDate
    )
    {
        if (customer is null)
        {
            throw new ArgumentNullException(nameof(customer));
        }

        if (equipment is null)
        {
            throw new ArgumentNullException(nameof(equipment));
        }

        EnsureValidDateRange(startDate, endDate);

        Id = Guid.NewGuid().ToString();
        Customer = customer;
        Equipment = equipment;
        StartDate = startDate;
        EndDate = endDate;
        Fee = CalculateFee(startDate, endDate);
    }

    public string Id { get; }

    public Customer Customer { get; }

    public Equipment Equipment { get; }

    public DateOnly StartDate { get; }

    public DateOnly EndDate { get; }

    public decimal Fee { get; }

    public static Rental? FindById(string id)
    {
        return Rentals.TryGetValue(id, out var rental) ? rental : null;
    }

    public static Rental Create(
        Customer customer,
        Equipment equipment,
        DateOnly startDate,
        DateOnly endDate
    )
    {
        if (customer is null)
        {
            throw new ArgumentNullException(nameof(customer));
        }

        if (equipment is null)
        {
            throw new ArgumentNullException(nameof(equipment));
        }

        EnsureValidDateRange(startDate, endDate);

        equipment.Reserve();

        var rental = new Rental(customer, equipment, startDate, endDate);

        Rentals.TryAdd(rental.Id, rental);

        customer.AddRental(rental);

        return rental;
    }

    private static void EnsureValidDateRange(DateOnly startDate, DateOnly endDate)
    {
        if (endDate <= startDate)
        {
            throw new ArgumentException("End date must be after start date.");
        }

        if (startDate <= DateOnly.FromDateTime(DateTime.UtcNow))
        {
            throw new ArgumentException("Start date must be in the future.");
        }
    }

    private static decimal CalculateFee(DateOnly startDate, DateOnly endDate)
    {
        var days = endDate.DayNumber - startDate.DayNumber;
        return days * CostPerDay;
    }

    public override string ToString()
    {
        return $"Rental {Id}: " +
               $"customer={Customer.Name}, " +
               $"equipment={Equipment.Name}, " +
               $"start={StartDate:yyyy-MM-dd}, " +
               $"end={EndDate:yyyy-MM-dd}, " +
               $"fee={Fee:C}";
    }
}
```

## Service

```csharp title="Services/RentalService.cs"
using RentalDddDemo.Domain;

namespace RentalDddDemo.Services;

public sealed class RentalService
{
    public Rental CreateRental(
        string customerId,
        string equipmentId,
        DateOnly startDate,
        DateOnly endDate
    )
    {
        var customer = Customer.FindById(customerId)
            ?? throw new KeyNotFoundException("Customer was not found.");

        var equipment = Equipment.FindById(equipmentId)
            ?? throw new KeyNotFoundException("Equipment was not found.");

        return Rental.Create(customer, equipment, startDate, endDate);
    }
}
```

The service no longer calculates fees or toggles availability — it resolves entities and asks `Rental.Create`.
