## Program.cs

```csharp title="Program.cs"
using RentalDddDemo.Domain;
using RentalDddDemo.Services;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddSingleton<RentalService>();

var app = builder.Build();

app.MapPost("/rentals", (CreateRentalRequest request, RentalService service) =>
{
    try
    {
        var rental = service.CreateRental(
            request.CustomerId,
            request.EquipmentId,
            request.StartDate,
            request.EndDate
        );

        return Results.Created($"/rentals/{rental.Id}", rental.ToResponse());
    }
    catch (KeyNotFoundException ex)
    {
        return Results.NotFound(new { error = ex.Message });
    }
    catch (ArgumentException ex)
    {
        return Results.BadRequest(new { error = ex.Message });
    }
    catch (InvalidOperationException ex)
    {
        return Results.Conflict(new { error = ex.Message });
    }
});

app.MapGet("/rentals/{id}", (string id) =>
{
    var rental = Rental.FindById(id);

    return rental is null
        ? Results.NotFound()
        : Results.Ok(rental.ToResponse());
});

app.Run();

public record CreateRentalRequest(
    string CustomerId,
    string EquipmentId,
    DateOnly StartDate,
    DateOnly EndDate
);

public record RentalResponse(
    string Id,
    string CustomerId,
    string CustomerName,
    string EquipmentId,
    string EquipmentName,
    DateOnly StartDate,
    DateOnly EndDate,
    decimal Fee
);

public static class RentalMappingExtensions
{
    public static RentalResponse ToResponse(this Rental rental)
    {
        return new RentalResponse(
            rental.Id,
            rental.Customer.Id,
            rental.Customer.Name,
            rental.Equipment.Id,
            rental.Equipment.Name,
            rental.StartDate,
            rental.EndDate,
            rental.Fee
        );
    }
}
```

## Run

```bash
dotnet run
```

Create a rental:

```bash
curl -X POST http://localhost:5000/rentals \
  -H "Content-Type: application/json" \
  -d '{"customerId":"customer-1","equipmentId":"equipment-1","startDate":"2026-09-10","endDate":"2026-09-12"}'
```

## Rule checks

| Case | Expect |
| --- | --- |
| Unknown customer / equipment | `404` |
| `endDate` ≤ `startDate` | `400` |
| Start in the past | `400` |
| Equipment already reserved | `409` |

Map exceptions at the edge; keep the **decision** in the domain.
