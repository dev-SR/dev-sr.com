## Program.cs

```csharp title="Program.cs"
using RentalDddDemo.Domain;
using RentalDddDemo.Services;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddSingleton<RentalService>();

var app = builder.Build();

app.MapPost("/rentals", (CreateRentalRequest request, RentalService service) =>
{
    var result = service.CreateRental(
        request.CustomerId,
        request.EquipmentId,
        request.StartDate,
        request.EndDate
    );

    if (result.IsFailure)
    {
        return result.ToFailureHttpResult();
    }

    return Results.Created(
        $"/rentals/{result.Value.Id}",
        result.Value.ToResponse()
    );
});

app.MapGet("/rentals/{id}", (string id) =>
{
    var result = Rental.FindById(id);

    if (result.IsFailure)
    {
        return result.ToFailureHttpResult();
    }

    return Results.Ok(result.Value.ToResponse());
});

app.Run();

public record CreateRentalRequest(
    string CustomerId,
    string EquipmentId,
    DateOnly StartDate,
    DateOnly EndDate
);

public record RentalResponse(
    string Id,
    string CustomerId,
    string CustomerName,
    string EquipmentId,
    string EquipmentName,
    DateOnly StartDate,
    DateOnly EndDate,
    decimal FeeAmount,
    string FeeCurrency
);

public static class ResultHttpExtensions
{
    public static IResult ToFailureHttpResult<T>(this Result<T> result)
    {
        var payload = new
        {
            error = result.Error
        };

        return result.ErrorType switch
        {
            ResultErrorType.NotFound => Results.NotFound(payload),
            ResultErrorType.Conflict => Results.Conflict(payload),
            _ => Results.BadRequest(payload)
        };
    }
}

public static class RentalMappingExtensions
{
    public static RentalResponse ToResponse(this Rental rental)
    {
        return new RentalResponse(
            rental.Id,
            rental.Customer.Id,
            rental.Customer.Name,
            rental.Equipment.Id,
            rental.Equipment.Name,
            rental.RentalPeriod.StartDate,
            rental.RentalPeriod.EndDate,
            rental.Fee.Amount,
            rental.Fee.Currency
        );
    }
}
```

## Happy path

```bash
curl -X POST http://localhost:5000/rentals \
  -H "Content-Type: application/json" \
  -d '{"customerId":"customer-1","equipmentId":"equipment-1","startDate":"2026-09-10","endDate":"2026-09-12"}'
```

## Failures to try

| Input | Domain signal |
| --- | --- |
| Bad customer id | `NotFound` |
| Bad equipment id | `NotFound` |
| Inverted dates | `Validation` |
| Past start | `Validation` |
| Second reserve on same gear | `Conflict` |

Map `ResultErrorType` once at the HTTP edge — do not re-validate in the controller.
